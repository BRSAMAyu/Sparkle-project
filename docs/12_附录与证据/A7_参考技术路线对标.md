## 一、学习软件范式：内容中心 vs 过程闭环

### 传统学习软件

```
┌─────────────────────────────────────────────────────────────┐
│  传统模式：内容/题库/课程为中心                              │
├─────────────────────────────────────────────────────────────┤
│  用户路径：浏览课程 → 观看视频 → 做题 → 查看分数            │
│                                                             │
│  核心问题：                                                  │
│  - 被动接收内容，缺乏个性化                                  │
│  - 学习过程不可追踪，无法针对性改进                          │
│  - 内容与用户状态割裂，难以形成闭环                          │
│  - 静态知识体系，无法适应用户认知水平                        │
└─────────────────────────────────────────────────────────────┘
```

**典型代表**: 网易云课堂、慕课网、刷题App

### Sparkle：过程与行为闭环为中心

```
┌─────────────────────────────────────────────────────────────┐
│  Sparkle模式：行为追踪 → 认知建模 → 自适应优化             │
├─────────────────────────────────────────────────────────────┤
│  用户路径：                                                  │
│  1. 行为追踪（任务完成、专注时长、学习模式）                 │
│  2. 认知建模（掌握度、薄弱点、学习偏好）                     │
│  3. 自适应优化（任务推荐、复习调度、难度调整）               │
│  4. 闭环反馈（效果评估 → 模型更新 → 下轮优化）              │
│                                                             │
│  核心优势：                                                  │
│  - 主动式学习：AI根据用户状态主动推送任务                   │
│  - 全链路追踪：从意图到执行的每个环节都可观测               │
│  - 动态知识图谱：学习进度实时可视化                         │
│  - 认知科学融入：间隔重复、主动回忆等科学方法               │
└─────────────────────────────────────────────────────────────┘
```

#### 3.1 GraphRAG检索器实现

```python
# backend/app/orchestration/graph_rag.py
class GraphRAGRetriever:
    """GraphRAG检索器"""

    async def retrieve(self, query: str, user_id: str) -> GraphRAGResult:
        """混合检索"""

        # 1. 向量检索
        vector_results = await self._vector_search(query, top_k=10)

        # 2. 图遍历
        graph_results = []
        for node in vector_results[:3]:  # 对Top-3节点进行图扩展
            # 获取相关节点（深度=2）
            related = await self._graph_traverse(
                node_id=node["id"],
                max_depth=2,
                min_strength=0.3
            )
            graph_results.extend(related)

        # 3. RRF融合
        fused_results = self._reciprocal_rank_fusion(
            [vector_results, graph_results],
            k=60
        )

        # 4. 重排序
        reranked = await self._rerank(query, fused_results)

        # 5. 构建上下文（包含关系路径）
        context = self._build_context_with_paths(
            reranked,
            include_relations=True
        )

        return GraphRAGResult(
            query=query,
            entities=[n["name"] for n in reranked],
            vector_results=vector_results,
            graph_results=graph_results,
            fused_context=context
        )
```

**图遍历实现**:
```python
async def _graph_traverse(
    self,
    node_id: str,
    max_depth: int = 2,
    min_strength: float = 0.3
) -> list[dict]:
    """图遍历：BFS获取相关节点"""

    visited = set()
    queue = [(node_id, 0)]  # (node_id, depth)
    results = []

    while queue:
        current_id, depth = queue.pop(0)

        if depth > max_depth:
            continue
        if current_id in visited:
            continue

        visited.add(current_id)

        # 获取相关节点
        relations = await self.db.execute(
            select(NodeRelation)
            .where(
                and_(
                    NodeRelation.source_id == current_id,
                    NodeRelation.strength >= min_strength
                )
            )
        )

        for relation in relations:
            target_node = await self.db.get(KnowledgeNode, relation.target_id)

            results.append({
                "id": target_node.id,
                "name": target_node.name,
                "relation_type": relation.relation_type,
                "strength": relation.strength,
                "depth": depth + 1
            })

            queue.append((target_node.id, depth + 1))

    return results
```

#### 3.3 个性化检索

**结合用户掌握度**:
```python
async def _personalized_rerank(
    self,
    user_id: str,
    candidates: list[dict]
) -> list[dict]:
    """个性化重排序"""

    # 获取用户掌握状态
    user_statuses = await self.db.execute(
        select(UserNodeStatus)
        .where(
            and_(
                UserNodeStatus.user_id == user_id,
                UserNodeStatus.node_id.in_([n["id"] for n in candidates])
            )
        )
    )

    status_map = {s.node_id: s for s in user_statuses}

    # 计算个性化分数
    for node in candidates:
        status = status_map.get(node["id"])

        if status:
            # 未掌握的节点优先
            if status.mastery_score < 60:
                node["personal_score"] = node["score"] * 1.5
            # 需要复习的节点次优先
            elif status.next_review_at < datetime.now():
                node["personal_score"] = node["score"] * 1.2
            else:
                node["personal_score"] = node["score"]
        else:
            # 未解锁的节点优先级高
            node["personal_score"] = node["score"] * 1.3

    # 按个性化分数排序
    return sorted(candidates, key=lambda x: x["personal_score"], reverse=True)
```


```python
# backend/app/orchestration/orchestrator.py
class ChatOrchestrator:
    """对话编排器"""

    def __init__(self):
        self.graph = self._build_workflow()
        self.checkpointer = RedisCheckpointer()
        self.context_pruner = ContextPruner()

    def _build_workflow(self) -> StateGraph:
        """构建LangGraph工作流"""
        workflow = StateGraph(WorkflowState)

        # 添加节点
        workflow.add_node("context_builder", self._context_builder_node)
        workflow.add_node("retrieval", self._retrieval_node)
        workflow.add_node("router", self._router_node)
        workflow.add_node("collaboration", self._collaboration_node)
        workflow.add_node("generation", self._generation_node)
        workflow.add_node("tool_execution", self._tool_execution_node)

        # 定义状态转换
        workflow.add_edge("context_builder", "retrieval")
        workflow.add_conditional_edges(
            "router",
            self._route_decision,
            {
                "direct": "generation",
                "langgraph": "collaboration",
                "hybrid": "tool_execution"
            }
        )

        return workflow.compile(checkpointer=self.checkpointer)

    async def _route_decision(self, state: WorkflowState) -> str:
        """智能路由决策"""
        intent = state["intent"]

        # 简单问答：直接生成
        if intent.type in ["greeting", "chitchat"]:
            return "direct"

        # 复杂任务：多Agent协作
        if intent.type in ["planning", "analysis", "reasoning"]:
            return "langgraph"

        # 工具调用：执行工具
        if intent.requires_tool:
            return "hybrid"
```

```python
# backend/app/orchestration/context_pruner.py
class ContextPruner:
    """上下文剪裁器"""

    MAX_CONTEXT_TOKENS = 4000
    RECENT_MESSAGES = 10
    SUMMARIZE_THRESHOLD = 20

    async def prune_context(
        self,
        messages: list[ChatMessage],
        system_prompt: str
    ) -> str:
        """智能剪裁上下文"""

        # 1. 检查Token数量
        total_tokens = sum(len(m.content) for m in messages)
        if total_tokens < self.MAX_CONTEXT_TOKENS:
            return self._build_context(messages, system_prompt)

        # 2. 超过阈值：总结历史
        if len(messages) > self.SUMMARIZE_THRESHOLD:
            recent = messages[-self.RECENT_MESSAGES:]
            history_to_summarize = messages[:-self.RECENT_MESSAGES]

            # LLM总结
            summary = await self._summarize_history(history_to_summarize)

            # 构建新上下文
            return self._build_context(
                recent,
                system_prompt + f"\n\n历史对话总结：\n{summary}"
            )

        # 3. 仅保留最近消息
        return self._build_context(
            messages[-self.RECENT_MESSAGES:],
            system_prompt
        )
```

```python
# backend/app/orchestration/dynamic_tool_registry.py
class DynamicToolRegistry:
    """动态工具注册表"""

    def __init__(self):
        self._tools: dict[str, BaseTool] = {}
        self._categories: dict[ToolCategory, list[str]] = {
            ToolCategory.TASK: [],
            ToolCategory.PLAN: [],
            ToolCategory.KNOWLEDGE: [],
            ToolCategory.QUERY: [],
            ToolCategory.FOCUS: [],
        }

    def register_tool(self, tool: BaseTool) -> None:
        """注册单个工具"""
        self._tools[tool.name] = tool
        self._categories[tool.category].append(tool.name)

    def register_from_package(self, package_name: str) -> None:
        """从包递归发现所有工具"""
        for module in import_submodules(package_name):
            for cls in module.__dict__.values():
                if isinstance(cls, type) and issubclass(cls, BaseTool):
                    self.register_tool(cls())

    def get_tools_for_category(
        self,
        category: ToolCategory
    ) -> list[BaseTool]:
        """获取指定类别的工具"""
        tool_names = self._categories.get(category, [])
        return [self._tools[name] for name in tool_names]
```

```python
class BaseTool(ABC):
    """工具基类"""

    name: str                          # 工具名称
    description: str                   # LLM理解的功能描述
```

```python
# backend/app/orchestration/executor.py
class ToolExecutor:
    """工具执行器"""

    MAX_PARAMS_SIZE = 10_000  # 10KB
    DEFAULT_TIMEOUT = 10  # 秒

    async def execute_tool(
        self,
        tool_name: str,
        parameters: dict
    ) -> ToolResult:
        """安全执行工具"""

        # 1. 获取工具
        tool = self.registry.get_tool(tool_name)
        if not tool:
            return ToolResult(
                success=False,
                error_message=f"Tool {tool_name} not found"
            )

        # 2. 参数验证
        try:
            validated_params = tool.parameters_schema(**parameters)
        except ValidationError as e:
            return ToolResult(
                success=False,
                error_message=f"Invalid parameters: {e}"
            )

        # 3. 检查参数大小
        params_size = len(json.dumps(parameters))
        if params_size > self.MAX_PARAMS_SIZE:
            return ToolResult(
                success=False,
                error_message="Parameters too large"
            )

        # 4. 执行工具（带超时）
        try:
            result = await asyncio.wait_for(
                tool.execute(**validated_params.dict()),
                timeout=self.DEFAULT_TIMEOUT
            )
        except asyncio.TimeoutError:
            return ToolResult(
                success=False,
                error_message="Tool execution timed out"
            )

        # 5. 记录执行历史
        await self._log_execution(tool_name, parameters, result)

        return result
```

| 维度 | 传统方案 | Sparkle方案 | 创新点 |
