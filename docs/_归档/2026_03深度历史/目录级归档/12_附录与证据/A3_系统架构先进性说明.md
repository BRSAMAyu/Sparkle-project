### 整体架构设计理念

Sparkle采用**三层混合微服务架构**，每一层使用最适合的技术栈，实现关注点分离和性能优化：

```
┌─────────────────────────────────────────────────────────────────┐
│  FLUTTER (Presentation)  →  轻推理、强交互、离线优先            │
├─────────────────────────────────────────────────────────────────┤
│  GO GATEWAY (Coordination)  →  鉴权、限流、长连接、会话治理      │
├─────────────────────────────────────────────────────────────────┤
│  PYTHON ENGINE (Intelligence)  →  智能编排、工具系统、检索分析   │
└─────────────────────────────────────────────────────────────────┘
         ↓ PostgreSQL+pgvector    ↓ Redis Stack    ↓ MinIO
```

```go
func (c *AgentClient) StreamChat(
    ctx context.Context,
    req *agentv1.ChatRequest,
) (<-chan *agentv1.ChatResponse, error) {
    stream, err := c.client.StreamChat(ctx, req)
    if err != nil {
        return nil, err
    }

    respChan := make(chan *agentv1.ChatResponse, 10)
    go func() {
        defer close(respChan)
        for {
            resp, err := stream.Recv()
            if err == io.EOF {
                return
            }
            if err != nil {
                logError(err)
                return
            }
            respChan <- resp
        }
    }()

    return respChan, nil
}
```

```python
async def idempotency_middleware(request: Request, call_next):
    # 1. 提取幂等键
    idempotency_key = request.headers.get("Idempotency-Key")
    if not idempotency_key:
        return await call_next(request)

    # 2. 检查是否已处理
    cached = await db.get(IdempotencyKey, idempotency_key)
    if cached:
        return JSONResponse(content=cached.response_json)

    # 3. 执行请求
    response = await call_next(request)

    # 4. 缓存响应
    await db.create(IdempotencyKey(
        id=idempotency_key,
        user_id=request.state.user_id,
        response_json=response.body,
        expires_at=datetime.utcnow() + timedelta(hours=24)
    ))

    return response
```

### 二、降级策略：失败时提供可用响应

#### 2.1 熔断器模式

**熔断器实现** (`backend/app/services/circuit_breaker.py`):
```python
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # 正常
    OPEN = "open"          # 熔断
    HALF_OPEN = "half_open"  # 半开（测试）

class CircuitBreaker:
    def __init__(self, failure_threshold=5, timeout=60):
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED
        self.failure_threshold = failure_threshold
        self.timeout = timeout

    async def call(self, func, *args, **kwargs):
        if self.state == CircuitState.OPEN:
            if self._should_attempt_reset():
                self.state = CircuitState.HALF_OPEN
            else:
                raise CircuitBreakerOpenError()

        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise

    def _should_attempt_reset(self) -> bool:
        return time.time() - self.last_failure_time >= self.timeout
```

**使用示例** (`backend/app/orchestration/orchestrator.py`):

```python
    async def search_with_fallback(self, query: str) -> list[Node]:

            # 尝试混合检索
            return await self.hybrid_search(query)
        except RedisError as e:
            logger.warning(f"Redis检索失败: {e}，降级到PG检索")
            try:
                return await self._pg_vector_search(query)
            except Exception as e:
                logger.error(f"PG检索也失败: {e}，返回空结果")
                return []
        except Exception as e:
            logger.error(f"检索失败: {e}，返回默认节点")
            return await self._get_default_nodes()
```
