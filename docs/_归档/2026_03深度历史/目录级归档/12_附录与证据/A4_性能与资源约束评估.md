### 弱网与移动端续航优化

#### 1. 长连接策略

**WebSocket心跳自适应**:
```dart
// backend/app/core/websocket.py
class WebSocketManager:
    PING_INTERVAL = 30  # 秒
    PING_TIMEOUT = 10

    async def start_ping(self, websocket):
        """自适应心跳：弱网时自动延长间隔"""
        while True:
            try:
                await asyncio.wait_for(
                    websocket.ping(),
                    timeout=self.PING_TIMEOUT
                )
                await asyncio.sleep(self.PING_INTERVAL)
            except TimeoutError:
                # 弱网检测：延长间隔到60秒
                await asyncio.sleep(60)
```

**连接复用**:
- 单一WebSocket连接处理聊天、文件、状态更新
- gRPC连接池复用（默认5个连接）
- HTTP/2多路复用

#### 4. 限流与配额治理

**多层限流**:
```go
// backend/gateway/internal/service/quota.go
type QuotaService struct {
    redis *redis.Client
}

// 用户日配额
func (s *QuotaService) CheckUserQuota(userID string) error {
    key := fmt.Sprintf("quota:user:%s:%s", userID, today)
    count, _ := s.redis.Get(ctx, key).Int()

    if count >= dailyLimit {
        return ErrQuotaExceeded
    }
    s.redis.Incr(ctx, key)
    s.redis.Expire(ctx, key, 24*time.Hour)
    return nil
}

// IP级限流（防刷）
func (s *QuotaService) CheckRateLimit(ip string) error {
    key := fmt.Sprintf("ratelimit:ip:%s", ip)
    // 滑动窗口：10次/分钟
    allowed, _ := s.redis.Allow(ctx, key, 10, time.Minute)
    if !allowed {
        return ErrRateLimitExceeded
    }
    return nil
}
```

**Token配额管理**:
```python
# backend/app/core/llm_quota.py
class LLMQuotaService:
    async def check_and_decr(
        self,
        user_id: str,
        estimated_tokens: int
    ) -> QuotaResult:
        # 1. 检查用户配额
        user_quota = await self.get_user_quota(user_id)

        if user_quota.remaining < estimated_tokens:
            return QuotaResult(allowed=False, reason="quota_exceeded")

        # 2. 扣除配额
        await self.redis.decrby(
            f"quota:tokens:{user_id}",
            estimated_tokens
        )

        return QuotaResult(allowed=True, remaining=user_quota - estimated_tokens)
```

## A4.2 成本控制策略（对AI项目很加分）

### 1. 语义缓存（相似问法复用）

**两层语义缓存**:

**L1: 精确文本匹配**
```go
// backend/gateway/internal/service/semantic_cache.go
func (s *SemanticCacheService) Canonicalize(input string) string {
    // "  如何  重置  密码？  " → "如何重置密码"
    sStr := strings.TrimSpace(strings.ToLower(input))
    sStr = strings.TrimRight(sStr, "?.!。？！")
    return sStr
}
```

**L2: 向量相似度搜索**
```go
func (s *SemanticCacheService) Search(
    ctx context.Context,
    vector []float32
) (string, error) {
    // KNN搜索（余弦相似度 > 0.85视为命中）
    cmd := s.rdb.Do(ctx,
        "FT.SEARCH", "idx:embeddings",
        "*=>[KNN 1 @vector $blob AS score]",
        "PARAMS", 2, "blob", blob,
        "SORTBY", "score",
    )
    // 返回最相似的历史问答
}
```

**缓存命中率优化**:
- TTL: 1小时（高频问法）
- 预热: 常见问题预先缓存
- 命中率监控: Prometheus指标记录

**成本节省估算**:

```python
class LLMService:

        "deepseek-chat",      # 首选（低成本）
```
