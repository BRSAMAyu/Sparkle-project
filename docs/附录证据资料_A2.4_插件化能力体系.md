**Step 2: 实现工具类**

```python
# 文件: backend/app/tools/web_search_tool.py

from .base import BaseTool, ToolCategory, ToolResult
from .schemas import WebSearchParams

class WebSearchTool(BaseTool):
    """Web 搜索工具"""
    name = "web_search"
    description = """在互联网上搜索最新信息。
    当用户询问时事、新闻、技术趋势或需要最新资料时使用，例如：
    - "搜索最新的 AI 进展"
    - "查找 Python 3.12 的新特性"
    - "搜一下今天的科技新闻"
    """
    category = ToolCategory.QUERY  # 或根据业务选择其他类别
    parameters_schema = WebSearchParams
    requires_confirmation = False  # 是否需要用户确认

    async def execute(
        self,
        params: WebSearchParams,
        user_id: str,
        db_session: Any,
        tool_call_id: str | None = None
    ) -> ToolResult:
        """
        执行 Web 搜索

        Args:
            params: 经过验证的参数对象
            user_id: 当前用户 ID
            db_session: 数据库会话
            tool_call_id: 工具调用唯一 ID

        Returns:
            ToolResult: 统一格式的执行结果
        """
        try:
            # 1. 调用搜索服务 (示例)
            import httpx
            async with httpx.AsyncClient() as client:
                response = await client.get(
                    "https://api.search.example.com/search",
                    params={
                        "q": params.query,
                        "num": params.num_results,
                        "type": params.search_type
                    },
                    timeout=10.0
                )
                response.raise_for_status()
                search_results = response.json()

            # 2. 格式化结果
            results = [
                {
                    "title": item["title"],
                    "url": item["link"],
                    "snippet": item["snippet"],
                    "source": item.get("source", "unknown")
                }
                for item in search_results.get("items", [])
            ]

            # 3. 返回 ToolResult
            return ToolResult(
                success=True,
                tool_name=self.name,
                data={
                    "query": params.query,
                    "result_count": len(results)
                },
                widget_type="search_results",  # 前端渲染组件类型
                widget_data={
                    "results": results,
                    "tool_result_id": tool_call_id
                }
            )

        except httpx.TimeoutException:
            return ToolResult(
                success=False,
                tool_name=self.name,
                error_message="搜索超时，请稍后重试",
                error_type="timeout",
                suggestion="尝试简化搜索关键词或检查网络连接"
            )

        except Exception as e:
            return ToolResult(
                success=False,
                tool_name=self.name,
                error_message=str(e),
                error_type="unknown",
                suggestion="如果问题持续，请尝试其他搜索方式"
            )
```

**Step 4: LLM 自动获取工具 Schema**

```python
# 文件: backend/app/orchestration/orchestrator.py

async def _build_tools_schema(self) -> list[dict]:
    """
    构建 OpenAI Function Calling 格式的工具 Schema

    这个方法会在每次 LLM 调用时执行，
    自动获取所有已注册工具的 Schema
    """
    # 自动生成 OpenAI 兼容的 Schema
    tools_schema = dynamic_tool_registry.get_openai_tools_schema()

    # 打印调试信息
    logger.info(f"已加载 {len(tools_schema)} 个工具:")
    for tool_schema in tools_schema:
        tool_name = tool_schema['function']['name']
        tool_desc = tool_schema['function']['description'][:50] + "..."
        logger.info(f"  - {tool_name}: {tool_desc}")

    return tools_schema

# LLM 调用时传递工具
async def _call_llm_with_tools(self, messages: list[dict]):
    tools_schema = await self._build_tools_schema()

    response = await llm_service.chat(
        messages=messages,
        tools=tools_schema,  # ← LLM 看到所有工具
        tool_choice="auto"   # LLM 自动选择工具
    )

    return response
```

from typing import Any
from .base import BaseTool, ToolCategory, ToolResult
from pydantic import BaseModel, Field

class WeatherParams(BaseModel):
    city: str = Field(..., description="城市名称")
    unit: str = Field(default="celsius", description="温度单位")

class WeatherTool(BaseTool):
    """天气查询工具"""
    name = "get_weather"
    description = "查询指定城市的实时天气"
    category = ToolCategory.QUERY
    parameters_schema = WeatherParams
    requires_confirmation = False

    async def execute(
        self,
        params: WeatherParams,
        user_id: str,
        db_session: Any,
        tool_call_id: str | None = None
    ) -> ToolResult:
        # 模拟天气数据
        mock_data = {
            "北京": {"temp": 5, "condition": "晴", "humidity": 45},
            "上海": {"temp": 12, "condition": "多云", "humidity": 60},
            "深圳": {"temp": 22, "condition": "阴", "humidity": 75}
        }

        data = mock_data.get(params.city, {"temp": 20, "condition": "未知", "humidity": 50})

        return ToolResult(
            success=True,
            tool_name=self.name,
            data={"city": params.city},
            widget_type="weather_card",
            widget_data={
                "city": params.city,
                "temperature": data["temp"],
                "condition": data["condition"],
                "humidity": data["humidity"],
                "unit": params.unit
            }
        )

| 维度 | 传统开发 | Sparkle 体系 |
